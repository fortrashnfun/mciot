// factorial

                     PRESERVE8
                     THUMB
                     AREA |.text|, CODE, READONLY
                     EXPORT __main
__main
                     MOVS R6, #05
                     MOVS R4, R6
                     SUBS R4, R4, #1
LOOP
                     MOVS R7, R4
                     MULS R7, R6, R7
                     MOVS R6, R7
                     SUBS R4, R4, #1
                     BNE  LOOP
STOP                 B    STOP
                     END

//Swap 10 bytes 

                     PRESERVE8
                     THUMB
                     AREA |.text|, CODE, READONLY
                     EXPORT __main
__main
                     LDR   R0, =0x20000000
                     LDR   R1, =0x20000010
                     MOVS R2, #10
LOOP
                     LDRB R3, [R0]
                     LDRB R4, [R1]
                     STRB R4, [R0]
                     STRB R3, [R1]
                     ADDS R0, R0, #1
                     ADDS R1, R1, #1
                     SUBS R2, R2, #1
                     BNE  LOOP
STOP                 B    STOP
                     END
